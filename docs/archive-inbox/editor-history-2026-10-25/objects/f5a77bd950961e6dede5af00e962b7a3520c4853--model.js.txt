export function formIssues(draft) {
  const issues = []
  const requireText = (value, issue) => { if (!String(value ?? '').trim()) issues.push(issue) }
  requireText(draft.title, { step: 0, field: 'title', message: '给案件起个名字' })
  requireText(draft.background, { step: 0, field: 'background', message: '补充玩家看到的公共背景' })
  if (!Number.isSafeInteger(draft.interrogationLimitSeconds) || draft.interrogationLimitSeconds <= 0) issues.push({ step: 0, field: 'interrogationLimitSeconds', message: '设置大于 0 的询问时长' })
  if (draft.characters.length === 0) issues.push({ step: 1, field: 'add-character', message: '至少添加一位人物' })
  draft.characters.forEach((character, index) => {
    const name = character.name.trim() || `人物 ${index + 1}`
    requireText(character.name, { step: 1, field: 'name', characterId: character.id, message: `填写人物 ${index + 1} 的姓名` })
    requireText(character.profile, { step: 1, field: 'profile', characterId: character.id, message: `补充${name}的人物设定` })
    if (!presets.some((appearance) => appearance.id === character.appearanceId)) issues.push({ step: 1, field: 'appearanceId', characterId: character.id, message: `为${name}选择一个预置形象` })
  })
  if (!draft.characters.some((character) => character.id === draft.ending.culpritId)) issues.push({ step: 1, field: 'culpritId', message: '在人物编辑中选定一位案件答案' })
  requireText(draft.ending.reveal, { step: 2, field: 'reveal', message: '写下正确指认后的完整经过' })
  return issues
}

export function removePerson(draft, id) {
  const index = draft.characters.findIndex((character) => character.id === id)
  if (index < 0) return null
  draft.characters.splice(index, 1)
  if (draft.ending.culpritId === id) draft.ending.culpritId = ''
  return draft.characters[Math.min(index, draft.characters.length - 1)]?.id ?? null
}
