<script setup lang="ts">
import { computed, onUnmounted, ref } from 'vue'
import Transcript from '../../packages/ui/src/components/Transcript.vue'
import UiButton from '../../packages/ui/src/components/UiButton.vue'
import { feedbackLabel, initialFeedback, reduceFeedback, rms } from './model.mjs'
const state = ref(initialFeedback())
const remaining = ref(480)
const input = ref(0)
const output = ref(0)
const answer = ref('')
const fallback = ref<string | null>(null)
const reduced = ref(false)
const note = ref('离线模拟；波形由合成 PCM 计算，不连接麦克风或灵矽。')
let revision = 0
let phase = 0
let textTimer: ReturnType<typeof setInterval> | null = null
function event(type: string, extra = {}) { state.value = reduceFeedback(state.value, { epoch: 1, type, ...extra }) }
function clearTimer() { if (textTimer != null) clearInterval(textTimer); textTimer = null }
function reset() { clearTimer(); state.value = initialFeedback(); answer.value = ''; fallback.value = null; revision = 0 }
function partial() {
  revision++
  event('level', { active: true })
  event('partial', { id: 'u1', revision, text: revision === 1 ? '你把章' : '你把盒子交给谁了' })
  note.value = '模拟 partial 原位修正，不写历史；灵矽公开协议尚未证明支持此能力。'
}
function final() { event('level', { active: false }); event('final', { id: 'u1', text: '你把盒子交给谁了？' }); note.value = 'final 只提交一次。迟到 partial 和重复 final 不新增回合。' }
function respond() {
  clearTimer(); fallback.value = null; answer.value = ''; event('playback', { active: true }); event('level', { active: false })
  const text = '盒子是我交出去的，当时章还在。后来是谁拿的，我也想知道。'
  note.value = '角色按 100ms/字展开，不在 2.4 秒内赶完。蓝条仍是合成 PCM，未播放角色语音。'
  if (reduced.value) { answer.value = text; return }
  let count = 0
  textTimer = setInterval(() => { answer.value = Array.from(text).slice(0, ++count).join(''); if (count >= Array.from(text).length) clearTimer() }, 100)
}
function fallbackText() {
  reset(); final(); const text = '你把盒子交给谁了？'; fallback.value = ''
  note.value = '整句已经收到后，本地按70ms/字展开；这不是实时识别。'
  if (reduced.value) { fallback.value = null; return }
  let count = 0
  textTimer = setInterval(() => { fallback.value = Array.from(text).slice(0, ++count).join(''); if (count >= Array.from(text).length) { clearTimer(); fallback.value = null } }, 70)
}
function stop() { clearTimer(); fallback.value = null; event('stop'); note.value = '结束/切人/隐藏：停止计时与音量反馈，丢弃未定稿草稿。' }
function choose(label: string) {
  if (!state.value.active) reset()
  if (label === '等你开口') reset()
  if (label === '玩家讲话') partial()
  if (label === '思考中') { final(); event('playback', { active: false }) }
  if (label === '角色回答') respond()
  if (label === '同时开口') { event('level', { active: true }); event('playback', { active: true }); note.value = '输入/输出可同时 active；实际打断前不伪造输出已经停止。' }
}
const messages = computed(() => {
  const rows = state.value.history.map((item: {id: string; text: string; mine: boolean}) => ({ ...item, speaker: '你', text: fallback.value ?? item.text }))
  if (state.value.draft != null) rows.push({ ...state.value.draft, mine: true, speaker: '你 · 识别中' })
  if (answer.value) rows.push({ id: 'a1', speaker: '沈禾', mine: false, text: answer.value })
  return rows
})
const time = computed(() => Math.floor(remaining.value / 60).toString().padStart(2, '0') + ':' + (remaining.value % 60).toString().padStart(2, '0'))
const bars = (value: number) => Array.from({ length: 15 }, (_, index) => 3 + Math.min(28, value * 90) * (0.3 + 0.7 * Math.sin(index * 0.7) ** 2))
const meter = setInterval(() => {
  phase += 0.23
  const samples = Float32Array.from({ length: 256 }, (_, index) => Math.sin(index * 0.13) * (0.15 + 0.1 * Math.sin(phase)))
  input.value = state.value.active && state.value.inputActive ? rms(samples) : 0
  output.value = state.value.active && state.value.outputActive ? rms(samples) : 0
}, 100)
const budget = setInterval(() => { if (state.value.active && remaining.value > 0) remaining.value-- }, 1000)
function visibility() { if (document.hidden) stop() }
document.addEventListener('visibilitychange', visibility)
onUnmounted(() => { clearTimer(); clearInterval(meter); clearInterval(budget); document.removeEventListener('visibilitychange', visibility) })
</script>
<template>
  <div class="fixture">
    <aside><strong>语音反馈 · 交互提案</strong><span>离线事件 / 合成 PCM，不连接麦克风或灵矽</span></aside>
    <header><a href="#" @click.prevent="stop">← 结束调查</a><strong>LIE</strong><UiButton variant="secondary">最终指认</UiButton></header>
    <main>
      <section class="person"><small>空盒里的纪念章</small><h1>沈禾</h1><div class="portrait"><img src="/appearances/sunny-moustache-v1/listening.png" alt="沈禾"></div></section>
      <section class="conversation"><h2>对话</h2><Transcript :messages="messages" :animate="false" empty-text="" /></section>
    </main>
    <footer>
      <div class="remaining"><span>剩余询问</span><strong>{{ time }}</strong></div>
      <div class="feedback"><strong role="status" aria-live="polite">{{ feedbackLabel(state) }}</strong><div class="meters" aria-hidden="true"><span class="wave input"><i v-for="(height, index) in bars(input)" :key="index" :style="{ height: (reduced ? 3 : height) + 'px' }"></i></span><span class="wave output"><i v-for="(height, index) in bars(output)" :key="index" :style="{ height: (reduced ? 3 : height) + 'px' }"></i></span></div><small>{{ !state.active ? '麦克风已关闭' : state.inputActive && state.outputActive ? '麦克风 + 播放器' : state.inputActive ? '麦克风音量' : state.outputActive ? '播放音量' : state.waitingForReply ? '正在等待回复' : '麦克风已开启' }}</small></div>
      <UiButton :variant="state.active ? 'secondary' : 'primary'" @click="state.active ? stop() : reset()">{{ state.active ? '结束对话' : '开始对话' }}</UiButton>
    </footer>
    <section class="controls"><p>{{ note }}</p><div><button v-for="label in ['等你开口', '玩家讲话', '思考中', '角色回答', '同时开口']" :key="label" @click="choose(label)">{{ label }}</button><button @click="partial">partial 修正</button><button @click="final">提交 final</button><button @click="fallbackText">整句 fallback</button><label><input v-model="reduced" type="checkbox">减少动态效果</label></div><small>绿＝输入，蓝＝输出。此处是明确标注的合成音量演示，不是实际收音/播放验收。</small></section>
  </div>
</template>
